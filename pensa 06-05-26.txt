# Solicitar los valores al usuario fila por fila
print("Ingresa los 9 números del cuadrado mágico 3x3:")

matriz = []
for i in range(3):
    fila = []
    for j in range(3):
        num = int(input(f"Posición [{i}][{j}]: "))
        fila.append(num)
    matriz.append(fila)

suma = [0, 0, 0]   
suma1 = [0, 0, 0]  

for i in range(len(matriz)):
    for j in range(len(matriz[0])):
        suma[i] += matriz[i][j]
        suma1[j] += matriz[i][j]

diag1 = 0
diag2 = 0
for i in range(3):
    diag1 += matriz[i][i]
    diag2 += matriz[i][2 - i]

print("\nLa matriz original es:", matriz)
print("Vector de filas: ", suma)
print("Vector de columnas:", suma1)
print("Diagonal principal:", diag1)
print("Diagonal secundaria:", diag2)

constante = suma[0]

if es_magico:
    all(s == constante for s in suma) and
    all(s == constante for s in suma1) and
    diag1 == constante and
    diag2 == constante
    
if es_magico:
    print(f"\n ¡Es un cuadrado mágico! Constante mágica = {constante}")
else:
    print("\nNo es un cuadrado mágico.")
