#  EJERCICIO 1

print("=" * 50)
print("           EJERCICIO 1 — Array 1D")
print("=" * 50)

M = [11, 22, 33, 44]
X = 0
Y = 2
print(f"\nEstado inicial:")
print(f"  M = {M}")
print(f"  X = {X}, Y = {Y}")

print("\n--- Paso a paso ---")

M[X] = M[Y]
print(f"M[X] = M[Y]  →  M[{X}] = M[{Y}] = {M[X]}  →  M = {M}")

M[Y] = M[1]
print(f"M[Y] = M[1]  →  M[{Y}] = {M[1]}            →  M = {M}")

X = M[0] - 30
print(f"X = M[0] - 30  →  X = {M[0]} - 30 = {X}")

M[3] = X + Y
print(f"M[3] = X + Y  →  M[3] = {X} + {Y} = {M[3]}  →  M = {M}")

Y = M[3] - 10
print(f"Y = M[3] - 10  →  Y = {M[3]} - 10 = {Y}")

M[1] = Y
print(f"M[1] = Y  →  M[1] = {Y}              →  M = {M}")

print(f"\n resultado final:")
print(f"   X = {X}")
print(f"   Y = {Y}")
print(f"   M = {M}")

#  EJERCICIO 2

print("\n" + "=" * 50)
print("           EJERCICIO 2 — Matriz 2D")
print("=" * 50)

M = [[1, 3, 5],
     [2, 4, 6],
     [7, 8, 9]]
A = 2
B = 1

print(f"\nEstado inicial:")
for i, fila in enumerate(M):
    print(f"  M[{i}] = {fila}")
print(f"  A = {A}, B = {B}")

print("\n--- Paso a paso ---")

M[A][B] = 10
print(f"M[A][B] = 10  →  M[{A}][{B}] = 10  →  M[{A}] = {M[A]}")

M[0][2] = M[2][1] - 2
print(f"M[0][2] = M[2][1] - 2  →  M[0][2] = {M[2][1]} - 2 = {M[0][2]}  →  M[0] = {M[0]}")

A = M[0][2] + 1
print(f"A = M[0][2] + 1  →  A = {M[0][2]} + 1 = {A}")

M[1][0] = A + B
print(f"M[1][0] = A + B  →  M[1][0] = {A} + {B} = {M[1][0]}  →  M[1] = {M[1]}")

M[2][2] = M[1][0] * 2
print(f"M[2][2] = M[1][0] * 2  →  M[2][2] = {M[1][0]} * 2 = {M[2][2]}  →  M[2] = {M[2]}")

print(f"\n resultado final:")
print(f"   A = {A}")
print(f"   B = {B}")
print(f"   Matriz M:")
for i, fila in enumerate(M):
    print(f"     M[{i}] = {fila}")
